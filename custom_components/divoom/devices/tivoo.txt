# turn off the display
action: divoom.off
data:
  entry_id: YOUR_DIVOOM_DEVICE

# change brightness of the display
action: divoom.brightness
data:
  entry_id: YOUR_DIVOOM_DEVICE
  brightness: 75

# activate channel 'clock' in rainbow style with specific components
action: divoom.clock
data:
  entry_id: YOUR_DIVOOM_DEVICE
  clock: 1
  weather: false
  temp: false
  calendar: true
  color: [250, 0, 0]
