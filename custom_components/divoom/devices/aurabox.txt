# turn off the display
action: divoom.off
data:
  device: YOUR_DIVOOM_DEVICE

# change brightness of the display
action: divoom.brightness
data:
  device: YOUR_DIVOOM_DEVICE
  brightness: 75

# activate channel 'clock' with 12h format
action: divoom.clock
data:
  device: YOUR_DIVOOM_DEVICE
  clock: 0
  twentyfour: false

# activate channel 'clock' with 24h format
action: divoom.clock
data:
  device: YOUR_DIVOOM_DEVICE
  clock: 0
  twentyfour: true

# activate channel 'temperature' with °C
action: divoom.temperature
data:
  device: YOUR_DIVOOM_DEVICE
  value: '°C'

# activate channel 'temperature' with °F
action: divoom.temperature
data:
  device: YOUR_DIVOOM_DEVICE
  value: '°F'

# activate light mode with specific color
action: divoom.light
data:
  device: YOUR_DIVOOM_DEVICE
  brightness: 75
  color: [250, 0, 0]

# activate channel 'effects' with specific effect
action: divoom.effects
data:
  device: YOUR_DIVOOM_DEVICE
  number: 2

# activate channel 'visualization' with specific visualization
action: divoom.visualization
data:
  device: YOUR_DIVOOM_DEVICE
  number: 0

# activate an image or animation, depending of the number of frames found in the file
action: divoom.image
data:
  device: YOUR_DIVOOM_DEVICE
  file: 'ha16.gif'

# activate a scrolling text animation with the Divoom font
action: divoom.text
data:
  device: YOUR_DIVOOM_DEVICE
  text: 'Hi Divoom'
  font: 'divoom.ttf'

# update date and time
action: divoom.datetime
data:
  device: YOUR_DIVOOM_DEVICE
  value: '2024-12-31 18:30:00'

# change the play/pause state
action: divoom.playstate
data:
  device: YOUR_DIVOOM_DEVICE
  value: true

# change the volume
action: divoom.volume
data:
  device: YOUR_DIVOOM_DEVICE
  volume: 75

# activates the radio and sets the frequency to 100.3
action: divoom.radio
data:
  device: YOUR_DIVOOM_DEVICE
  value: true
  frequency: 100.3

# activates the alarm in the first slot for 7:30 in the morning on monday-friday
action: divoom.alarm
data:
  device: YOUR_DIVOOM_DEVICE
  number: 0
  value: '07:30'

# activate the sleep mode for 30 minutes
action: divoom.sleep
data:
  device: YOUR_DIVOOM_DEVICE
  value: true
  time: 30
  sleepmode: 4

# activates the memorial in the first slot for 0:00 new year
action: divoom.memorial
data:
  device: YOUR_DIVOOM_DEVICE
  number: 0
  value: '2000-12-31 00:00:00'
  text: 'Happy New Year!'

# sends a raw command which sets brightness to 100%
action: divoom.raw
data:
  device: YOUR_DIVOOM_DEVICE
  raw: [0x74, 0x64]
